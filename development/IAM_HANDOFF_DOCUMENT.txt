Add IAM version:
```fortran
function dtauda_IAM(a, beta)
  ! IAM Hubble parameter
  real(dl) :: a, beta, dtauda_IAM
  real(dl) :: E_act, H_squared
  
  ! Activation function
  E_act = exp(1.0_dl - 1.0_dl/a)
  
  ! IAM Friedmann equation
  H_squared = this%omegam * a**(-3) + &
              this%omegar * a**(-4) + &
              this%omegal + &
              beta * E_act
  
  ! Convert to conformal time derivative
  dtauda_IAM = sqrt(3.0_dl / H_squared) / (const_c * 1000._dl)
  
  return
end function dtauda_IAM
```

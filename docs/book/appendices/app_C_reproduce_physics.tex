\chapter{Reproducing every number and figure}\label{app:repro}

Every figure is produced by one script, \texttt{make\_figures.py}, shipped with the source of
this book. It uses only NumPy, SciPy and Matplotlib and the constants of
Appendix~\ref{app:constants}. The checks in the grey boxes are short computations. The
principal ones are reproduced below so that a reader can verify them in a few lines.

\begin{verbatim}
import math
hbar,c,G,kB = 1.054571817e-34, 2.99792458e8, 6.67430e-11, 1.380649e-23
# Ch. 16: x_qp with both Cooper-pair densities
tq, tT, lam = 100e-6, 30e-6, 50e-9;  V = math.pi*lam**3
for ncp in (9.03e28, 4e24):
    print(ncp, math.log(2)*tq/(ncp*tT*V))       # 6.5e-8 ; 1.5e-3
# Ch. 5: the cosmological-constant relation reduces to
print(math.sqrt(0.685), 16/3*0.0493/0.315)      # 0.828 ; 0.835
# Ch. 9: electron mass, prefactor (2pi)^(-1/10), H0 = 67.4
al=7.2973525693e-3; mP=math.sqrt(hbar*c/G); H0=67.4e3/3.0857e22
me=(2*math.pi)**-0.1*(hbar*H0*math.log(2)*mP**1.5/(al**2.5*c**2))**0.4
print(me/9.1093837015e-31)                       # 1.0000066
# Ch. 13: IAM gravitational decoherence time
EG=lambda m,R: G*m*m/R
tau=lambda m,R,T: hbar*kB**2*T**2*math.log(2)/EG(m,R)**3
R=(3e-12/(4*math.pi*2200))**(1/3); print(tau(1e-12,R,0.01))  # ~5e2 s
\end{verbatim}

The chains, extraction reports and source papers referred to in the text are in the
repository \url{https://github.com/hmahaffeyges/IAM-Validation}, under \texttt{docs/papers/}.

\chapter{Reproducing the physics numbers and figures}\label{app:repro}

The figures of Parts~\ref{part:1}, \ref{part:2}, \ref{part:3} and~\ref{part:5} are built by the scripts in \texttt{docs/\allowbreak book/\allowbreak figscripts/}
of the repository \url{https://github.com/hmahaffeyges/IAM-Validation}; each script writes its figures into \texttt{docs/\allowbreak book/\allowbreak figures/}.
They use NumPy, SciPy and Matplotlib and the constants of Appendices~\ref{app:constants} and~\ref{app:notation}. The Planck chains and their extraction
scripts are in \texttt{mgcamb\_validation/} (Level~1) and \texttt{camb\_validation/} (Level~2), and the check files and script outputs behind the numbers are in
\texttt{docs/\allowbreak verification/}. The principal numbers can be checked in a few lines; the comments name the chapters (Chapters~\ref{ch:virial},
\ref{ch:lambda}, \ref{ch:particle}, \ref{ch:gravdec}, \ref{ch:cmos} and~\ref{ch:blackholes}).

\begin{Verbatim}[breaklines,breakanywhere,fontsize=\footnotesize]
import math
hbar, c, G, kB = 1.054571817e-34, 2.99792458e8, 6.67430e-11, 1.380649e-23
Mpc = 648000/math.pi*149597870700*1e6           # metres (IAU parsec)
# The cosmological coupling: beta_m = Omega_m/2, mu_0 and the matter-sector H0
Om = 0.3153; b = Om/2
print(b, -b/(1+b), 67.16*math.sqrt(1+b))        # 0.15765 ; -0.136 ; 72.26
# The cosmological constant: Omega_b/Omega_m = (3/16) sqrt(Omega_Lambda)
print(math.sqrt(0.685), 16/3*0.0493/0.315)      # 0.828 ; 0.835
# The particle scale: electron mass, prefactor (2 pi)^(-1/10)
al = 7.2973525693e-3; mP = math.sqrt(hbar*c/G)
for h in (67.4, 67.36):
    H0 = h*1e3/Mpc
    me = (2*math.pi)**-0.1*(hbar*H0*math.log(2)*mP**1.5/(al**2.5*c**2))**0.4
    print(h, me/9.1093837015e-31)               # 1.0000066 ; 0.99977
# Gravitational decoherence: tau_IAM and tau_DP for 1e-12 kg of silica at 10 mK
EG = lambda m, R: G*m*m/R
tau = lambda m, R, T: hbar*kB**2*T**2*math.log(2)/EG(m, R)**3
R = (3e-12/(4*math.pi*2200))**(1/3)
print(tau(1e-12, R, 0.01), hbar/EG(1e-12, R))   # 509 s ; 7.5e-6 s
# Landauer at the transistor gate: the floor for logic at T_j = 348.15 K
print(kB*348.15*math.log(2), kB*348.15*math.log(2)/1.602176634e-19)   # 3.33e-21 J ; 0.021 eV
# Black-hole horizons: Kerr fraction T S / M c^2 at spin 0.9
print(math.sqrt(1 - 0.9**2)/2)                  # 0.218
\end{Verbatim}

\noindent The electron mass scales as $H_0^{2/5}$, so the $0.8\,\%$ uncertainty of $H_0$ sets its $\pm0.3\,\%$ precision; between $H_0=67.4$ and $67.36$ it moves by
$0.024\,\%$.
